public class divisible{
    public static void main(String[] args){
        int a = 2;
        int b = 15;
        int c = 3;
        int count = 0;
        for (int i = a ; i<=b ; i++){
            if (i % c == 0){
                count++;
            }
        }
        System.out.println(count);
    }
}
